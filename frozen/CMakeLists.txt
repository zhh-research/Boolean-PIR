set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(libOTe CONFIG REQUIRED)

target_include_directories(pbm_frozen PRIVATE generated)
target_link_libraries(pbm_frozen PRIVATE oc::libOTe)
target_compile_options(pbm_frozen PRIVATE -Wall -Wextra)
